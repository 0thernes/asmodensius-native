# gpu/CMakeLists.txt - the GPU path. Added by the main build when the
# component "gpu" is selected (-DAZ_COMPONENTS="gpu" or ALL).
#
# Builds:
#   az_gpu_dev.ptx   device code: clang --target=nvptx64-nvidia-cuda, PTX text
#   az_gpu           static library: driver-API host + the PTX embedded as bytes
#   az_gpu_ref       the CPU reference: the same two C files, compiled by the
#                    host compiler with the flags az_core uses
#   test_gpu         GPU == CPU bit for bit (exit 0 with a message if no driver)
#   az_gpubench      timings of the brain-step batches
#
# No CUDA toolkit is used. If clang is not installed the library is built
# without device code and az_gpu_open reports AZ_GPU_NO_PTX.
#
# The component links nothing from az_core, az_host or az_kernels, so another
# module that does not compile at the moment cannot break it.

set(AZ_GPU_CLANG "Z:/Tools/LLVM-21/bin/clang.exe" CACHE FILEPATH
    "clang with the nvptx64 target; the device code is skipped when this file does not exist")
set(AZ_GPU_ARCH "sm_80" CACHE STRING "-march of the device code (sm_80 gives PTX 7.0)")
set(AZ_GPU_BRAIN_SRC "" CACHE FILEPATH "brain source for device and reference; empty = core/az_brain.c, else the stub")
set(AZ_GPU_RNG_SRC "" CACHE FILEPATH "Philox source for device and reference; empty = core/az_rng.c, else the stub")

set(AZ_GPU_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(AZ_GPU_BIN "${CMAKE_CURRENT_BINARY_DIR}")
get_filename_component(AZ_GPU_ROOT "${CMAKE_SOURCE_DIR}" ABSOLUTE)
string(REGEX REPLACE "/+$" "" AZ_GPU_ROOT "${AZ_GPU_ROOT}")

# One definition of the semantics: device code and CPU reference are both
# compiled from core/az_<m>.c when that file exists, else from the stub.
foreach(m rng)
  string(TOUPPER "${m}" M)
  if(AZ_GPU_${M}_SRC)
    set(az_gpu_src_${m} "${AZ_GPU_${M}_SRC}")
  elseif(EXISTS "${AZ_GPU_ROOT}/core/az_${m}.c")
    set(az_gpu_src_${m} "${AZ_GPU_ROOT}/core/az_${m}.c")
  else()
    set(az_gpu_src_${m} "${AZ_GPU_ROOT}/core/stubs/az_${m}.c")
  endif()
endforeach()
# The batch network of the GPU twin is the dense one (kernels/az_dense.c). The
# engine's per-creature brain (core/az_brain.c) has no device port yet.
if(AZ_GPU_BRAIN_SRC)
  set(az_gpu_src_brain "${AZ_GPU_BRAIN_SRC}")
else()
  set(az_gpu_src_brain "${AZ_GPU_ROOT}/kernels/az_dense.c")
endif()
file(CONFIGURE OUTPUT "${AZ_GPU_BIN}/az_gpu_dev_srcs.h" CONTENT
"/* generated by gpu/CMakeLists.txt: the core sources shared by the device code
 * (az_gpu_dev.c) and the CPU reference (az_gpu_ref.c) */
#include \"${az_gpu_src_rng}\"
#include \"${az_gpu_src_brain}\"
" @ONLY NEWLINE_STYLE LF)
message(STATUS "az gpu: shared sources ${az_gpu_src_rng} ${az_gpu_src_brain}")
set(AZ_GPU_SHARED_DEPS
    "${AZ_GPU_BIN}/az_gpu_dev_srcs.h" "${az_gpu_src_rng}" "${az_gpu_src_brain}"
    "${AZ_GPU_ROOT}/core/az_abi.h" "${AZ_GPU_ROOT}/core/az_types.h"
    "${AZ_GPU_ROOT}/kernels/az_dense.h"
    "${AZ_GPU_ROOT}/core/az_columns.def" "${AZ_GPU_ROOT}/core/az_cells.def")

# ---- device code -> PTX -> C byte array ------------------------------------
set(AZ_GPU_LIB_SRCS "${AZ_GPU_DIR}/az_gpu.c")
set(AZ_GPU_DEFS "")
if(EXISTS "${AZ_GPU_CLANG}")
  set(az_gpu_ptx "${AZ_GPU_BIN}/az_gpu_dev.ptx")
  set(az_gpu_ptx_c "${AZ_GPU_BIN}/az_gpu_ptx.c")
  # -fwrapv and -fno-strict-aliasing: the same integer model az_core is built with.
  add_custom_command(
    OUTPUT "${az_gpu_ptx}"
    COMMAND "${AZ_GPU_CLANG}" --target=nvptx64-nvidia-cuda -march=${AZ_GPU_ARCH} -O2 -std=gnu23
            -ffreestanding -fwrapv -fno-strict-aliasing
            -Wall -Wextra -Wno-unused-function -Wno-unused-parameter -Wno-undefined-internal
            -I "${AZ_GPU_ROOT}/core" -I "${AZ_GPU_ROOT}/kernels" -I "${AZ_GPU_DIR}" -I "${AZ_GPU_BIN}"
            -MD -MF "${az_gpu_ptx}.d"
            -S "${AZ_GPU_DIR}/az_gpu_dev.c" -o "${az_gpu_ptx}"
    DEPENDS "${AZ_GPU_DIR}/az_gpu_dev.c" "${AZ_GPU_DIR}/az_gpu_layout.h" ${AZ_GPU_SHARED_DEPS}
    DEPFILE "${az_gpu_ptx}.d"
    COMMENT "clang nvptx64 (${AZ_GPU_ARCH}): az_gpu_dev.c -> az_gpu_dev.ptx"
    VERBATIM)
  # check the PTX (entries present, no unresolved function, no floating point) and embed it
  add_custom_command(
    OUTPUT "${az_gpu_ptx_c}"
    COMMAND "${CMAKE_COMMAND}" "-DIN=${az_gpu_ptx}" "-DOUT=${az_gpu_ptx_c}"
            -P "${AZ_GPU_DIR}/az_gpu_embed.cmake"
    DEPENDS "${az_gpu_ptx}" "${AZ_GPU_DIR}/az_gpu_embed.cmake"
    COMMENT "check and embed az_gpu_dev.ptx"
    VERBATIM)
  list(APPEND AZ_GPU_LIB_SRCS "${az_gpu_ptx_c}")
  list(APPEND AZ_GPU_DEFS AZ_GPU_HAVE_PTX=1)
  message(STATUS "az gpu: device code built with ${AZ_GPU_CLANG} for ${AZ_GPU_ARCH}")
else()
  message(STATUS "az gpu: ${AZ_GPU_CLANG} not found; building without device code")
endif()

# ---- host library ----------------------------------------------------------
add_library(az_gpu STATIC ${AZ_GPU_LIB_SRCS})
target_include_directories(az_gpu PUBLIC "${AZ_GPU_DIR}" "${AZ_GPU_ROOT}/core" "${AZ_GPU_ROOT}/kernels")
target_compile_definitions(az_gpu PRIVATE ${AZ_GPU_DEFS})
target_compile_options(az_gpu PRIVATE ${AZ_WARN} -O2 -march=${AZ_MARCH})

# ---- CPU reference: same sources, same flags as az_core ---------------------
# Each function in its own section, and the executables link with
# --gc-sections: only what a test reaches is kept, so world code in the brain
# file that calls into other modules does not have to resolve.
add_library(az_gpu_ref STATIC "${AZ_GPU_DIR}/az_gpu_ref.c")
target_include_directories(az_gpu_ref PUBLIC "${AZ_GPU_ROOT}/core" "${AZ_GPU_ROOT}/kernels" PRIVATE "${AZ_GPU_BIN}")
target_compile_options(az_gpu_ref PRIVATE ${AZ_WARN} -ffreestanding -fno-strict-aliasing -fwrapv
                       -march=${AZ_MARCH} -O2 -ffunction-sections)
set_source_files_properties("${AZ_GPU_DIR}/az_gpu_ref.c" PROPERTIES OBJECT_DEPENDS "${AZ_GPU_SHARED_DEPS}")

# ---- test and tool (they live in gpu/ and are built with the component) -----
foreach(exe test_gpu az_gpubench)
  add_executable(${exe} "${AZ_GPU_DIR}/${exe}.c")
  target_include_directories(${exe} PRIVATE "${AZ_GPU_ROOT}/tests")
  target_compile_options(${exe} PRIVATE ${AZ_WARN} -O2 -march=${AZ_MARCH})
  target_link_options(${exe} PRIVATE -static -Wl,--gc-sections)
  target_link_libraries(${exe} PRIVATE az_gpu az_gpu_ref)
  set_target_properties(${exe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
# Optional second CPU baseline for the benchmark: the AVX-VNNI assembly twin
# (another owner's file, so it is off unless asked for).
option(AZ_GPU_BENCH_TWIN "az_gpubench also times kernels/az_brain_vnni.asm" OFF)
if(AZ_GPU_BENCH_TWIN AND EXISTS "${AZ_GPU_ROOT}/kernels/az_brain_vnni.asm")
  find_program(AZ_GPU_NASM nasm)
  if(AZ_GPU_NASM)
    set(az_gpu_twin "${AZ_GPU_BIN}/az_brain_vnni.obj")
    add_custom_command(
      OUTPUT "${az_gpu_twin}"
      COMMAND "${AZ_GPU_NASM}" -f win64 "-I${AZ_GPU_ROOT}/kernels/"
              "${AZ_GPU_ROOT}/kernels/az_brain_vnni.asm" -o "${az_gpu_twin}"
      DEPENDS "${AZ_GPU_ROOT}/kernels/az_brain_vnni.asm" "${AZ_GPU_ROOT}/kernels/win64.inc"
      COMMENT "nasm: az_brain_vnni.asm (benchmark baseline)"
      VERBATIM)
    target_sources(az_gpubench PRIVATE "${az_gpu_twin}")
    set_source_files_properties("${az_gpu_twin}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
    target_compile_definitions(az_gpubench PRIVATE AZ_GPU_BENCH_VNNI=1)
  endif()
endif()

add_test(NAME test_gpu COMMAND test_gpu)
set_tests_properties(test_gpu PROPERTIES TIMEOUT 600)
